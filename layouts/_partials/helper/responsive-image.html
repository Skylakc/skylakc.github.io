{{- /* Params: Resource: Hugo image resource object (Optional, for processing) Widths: Slice of widths to generate for srcset Attributes: Map of HTML attributes for the tag Local override of the Stack helper: - Never offer an original wider than 1600px in srcset (the originals are often 4000-6000px and dominate page weight on high-DPI screens). - Add decoding="async" to every image. */ -}} {{- $resource := .Resource -}} {{- $widths := .Widths -}} {{- $attributes := .Attributes | default dict -}} {{- $maxOriginalWidth := 1600 -}} {{- if and $widths $resource (reflect.IsImageResourceProcessable $resource) -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $widths -}} {{- if lt . $resource.Width -}} {{- $resized := $resource.Resize (printf "%dx" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- $fallback = $resized.RelPermalink -}} {{- end -}} {{- end -}} {{- if and (gt (len $srcset) 0) (le $resource.Width $maxOriginalWidth) -}} {{- $permalink := default $resource.RelPermalink (index $attributes "src") -}} {{- $srcset = $srcset | append (printf "%s %dw" $permalink $resource.Width) -}} {{- end -}} {{- if gt (len $srcset) 0 -}} {{- $attributes = merge $attributes (dict "srcset" (delimit $srcset ", ")) -}} {{- if and $fallback (gt $resource.Width $maxOriginalWidth) -}} {{- $attributes = merge $attributes (dict "src" $fallback) -}} {{- end -}} {{- end -}} {{- end -}} {{- $attributes = merge $attributes (dict "decoding" "async") -}}
